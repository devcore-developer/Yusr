"use client"

import { useState } from "react"
import { Button } from "@/components/ui/Button"
import { Input } from "@/components/ui/Input"
import { Textarea } from "@/components/ui/Textarea"
import { addSurvey } from "@/features/landing/actions"
import { Loader2, CheckCircle } from "lucide-react"

interface SurveyFormProps {
  evidenceTexts: { tool: string; text: string }[]
}

export function SurveyForm({ evidenceTexts }: SurveyFormProps) {
  const [loading, setLoading] = useState(false)
  const [success, setSuccess] = useState(false)

  // تجميع الشواهد حسب الأداة
  const groupedEvidence = evidenceTexts.reduce((acc, item) => {
    if (!acc[item.tool]) acc[item.tool] = []
    acc[item.tool].push(item.text)
    return acc
  }, {} as Record<string, string[]>)

  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
    e.preventDefault()
    setLoading(true)
    setSuccess(false)
    
    const formData = new FormData(e.currentTarget)
    try {
      await addSurvey(formData)
      setSuccess(true)
      e.currentTarget.reset()
      setTimeout(() => setSuccess(false), 3000)
    } catch (error) {
      console.error(error)
    }
    setLoading(false)
  }

  return (
    <form onSubmit={handleSubmit} className="space-y-4">
      {success && (
        <div className="flex items-center gap-2 p-3 bg-success/10 text-success rounded-lg text-sm">
          <CheckCircle className="w-4 h-4" /> تم إضافة نتيجة الاستبانة بنجاح.
        </div>
      )}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
        <div className="space-y-1.5">
          <label className="text-sm font-medium">نوع الاستبانة</label>
          <select name="type" className="w-full h-10 px-3 rounded-md border border-border bg-surface text-sm">
            <option value="استبانة الطلاب">استبانة الطلاب</option>
            <option value="استبانة المعلمين">استبانة المعلمين</option>
            <option value="استبانة أولياء الأمور">استبانة أولياء الأمور</option>
          </select>
        </div>
        <div className="space-y-1.5">
          <label className="text-sm font-medium">النسبة المئوية %</label>
          <Input name="percentage" type="number" min="0" max="100" step="0.1" required placeholder="85.5" />
        </div>
        <div className="space-y-1.5">
          <label className="text-sm font-medium">تضمين في التقييم؟</label>
          <select name="include" className="w-full h-10 px-3 rounded-md border border-border bg-surface text-sm">
            <option value="false">لا (فقط للعرض)</option>
            <option value="true">نعم (تدخل في النسبة)</option>
          </select>
        </div>
      </div>

      {/* حقل النتيجة العامة للاستبانة */}
      <div className="space-y-1.5">
        <label className="text-sm font-medium">النتيجة العامة للاستبانة</label>

"use server"

import { prisma } from "@/lib/prisma"
import { getCurrentUser } from "@/lib/permissions"
import { revalidatePath } from "next/cache"

export async function submitContactMessage(formData: FormData) {
  const name = formData.get("name") as string
  const institution = formData.get("institution") as string
  const email = formData.get("email") as string
  const phone = formData.get("phone") as string
  const subject = formData.get("subject") as string
  const message = formData.get("message") as string

  if (!name || !email || !subject || !message) {
    return { error: "الرجاء تعبئة الحقول الإلزامية." }
  }

  try {
    await prisma.contactMessage.create({
      data: { name, institution: institution || null, email, phone: phone || null, subject, message },
    })
    return { success: true }
  } catch (error) {
    return { error: "حدث خطأ أثناء إرسال الرسالة. الرجاء المحاولة مرة أخرى." }
  }
}

export async function addSurvey(formData: FormData) {
  const user = await getCurrentUser()
  if (!user) throw new Error("Unauthorized")

  const type = formData.get("type") as string
  const percentage = parseFloat(formData.get("percentage") as string)
  const include = formData.get("include") === "true"
  const note = formData.get("note") as string
  const generalResult = formData.get("generalResult") as string

  if (!type || isNaN(percentage)) {
    return { error: "البيانات غير صحيحة" }
  }

  try {
    await prisma.survey.create({
      data: {
        type,
        percentage,
        include,
        note: note || null,
        generalResult: generalResult || null,
        schoolId: user.schoolId || null,
      }
    })
    
    revalidatePath("/surveys")
    return { success: true }
  } catch (error) {
    return { error: "حدث خطأ أثناء حفظ الاستبانة." }
  }
}

  })

    branchId: a.branchId,
    branchName: a.branch?.name || "كامل المدرسة"
  }))

  const provisionalPercentage = mapped.length > 0
    ? Math.round((mapped.reduce((sum, a) => sum + a.score, 0) / mapped.length) * 10) / 10

  const finalPercentage = completed.length > 0

    : null

  for (const a of mapped) {

        id: a.evaluatorId,
        name: a.evaluatorName,
        role: a.evaluatorRole,
        assessments: [],
        provisionalAverage: 0,
        finalAverage: null
      })
    }
    evaluatorMap.get(a.evaluatorId)!.assessments.push(a)
  }

  const evaluators: EvaluatorAnalytics[] = Array.from(evaluatorMap.values()).map(ev => {
    const evCompleted = ev.assessments.filter(a => a.status === "COMPLETED")
    ev.provisionalAverage = ev.assessments.length > 0
      ? Math.round((ev.assessments.reduce((sum, a) => sum + a.score, 0) / ev.assessments.length) * 10) / 10
      : null

  })

    provisionalPercentage,
    finalPercentage,
    eligibleAssessments: mapped.length,
    completedAssessments: completed.length,
    inProgressAssessments: mapped.length - completed.length,

  }

export function getPerformanceBand(score: number | null): string {
  if (score === null || Number.isNaN(score)) return "غير مقيم"

  if (score >= 75) return "جيد جدًا"

export interface ComparisonResult {

  indicatorText: string
  domain: string
  responses: Record<string, { value: string; notes?: string | null } | undefined>
  agreement: boolean
  evaluatedByAll: boolean

export interface AggregatedResult {
  totalEvaluations: number
  averageScore: number
  performanceBand: string
  domainAverages: { domain: string; average: number; band: string }[]
  agreementRate: number
  totalIndicators: number
  agreedIndicators: number
  disagreedIndicators: number
}

export async function compareAssessments(assessmentIds: string[]) {
  // الإصلاح: حراسة داخلية — أي مستدٍٍى لهذه الخدمة (صفحة المقارنة أو الطباعة)
  // يمر إلزامياً عبر فحص الوصول لكل تقييم مطلوب، فلا يمكن استغلالها من مسار منسي
  const currentUser = await getCurrentUser()
  if (!currentUser) throw new Error("غير مصرح")

  for (const id of assessmentIds) {
    const access = await checkAssessmentAccess(id, "read")
    if (!access.allowed) {
      throw new Error("غير مصرح بمقارنة أحد التقييمات المحددة")
    }
  }

  const assessments = await prisma.assessment.findMany({
    where: { id: { in: assessmentIds } },
    include: {
      assessor: { select: { id: true, name: true, role: true } },
      school: { select: { name: true } },
      branch: { select: { name: true } },
      evidenceResponses: {
        where: { isVerified: true },
        include: {
          evidence: {
            include: {
              indicator: {
                include: { domain: true, axis: true }
              }
            }
          }
        }
      },
      responses: {
        include: {
          indicator: true
        }
      }
    }
  })

  if (assessments.length < 2) {
    throw new Error("يجب اختيار تقييمين على الأقل للمقارنة")
  }

  const domains = await prisma.domain.findMany({
    include: {
      indicators: {
        include: { fixedEvidence: true, axis: true }
      }
    }
  })

  const allIndicators = domains.flatMap(d => 
    d.indicators.map(i => ({
      code: i.code,
      text: i.description,
      domain: d.name,
      axis: i.axis?.name,
      requiredEvidence: i.fixedEvidence.length
    }))
  )

  const comparisons: ComparisonResult[] = allIndicators.map(indicator => {
    const responses: Record<string, { value: string; notes?: string | null } | undefined> = {}
    let allEvaluated = true
    let allAgree = true
    let firstValue: string | null = null

    for (const assessment of assessments) {
      const verifiedCount = assessment.evidenceResponses.filter(
        er => er.evidence.indicator.code === indicator.code
      ).length

      const response = assessment.responses.find(
        r => r.indicator.code === indicator.code
      )

      const status = verifiedCount > 0 
        ? `${verifiedCount}/${indicator.requiredEvidence} متحقق`
        : "لم يُقيّم"

      responses[assessment.assessor?.id ?? "unknown"] = {
        value: status,
        notes: response?.notes
      }

      if (verifiedCount === 0) {
        allEvaluated = false
      }

      if (firstValue === null) {
        firstValue = status
      } else if (status !== firstValue) {
        allAgree = false
      }
    }

    return {
      indicatorCode: indicator.code,
      indicatorText: indicator.text,
      domain: indicator.domain,
      responses,
      agreement: allAgree && allEvaluated,
      evaluatedByAll: allEvaluated
    }
  })

  const totalIndicators = comparisons.length
  const evaluatedIndicators = comparisons.filter(c => c.evaluatedByAll).length
  const agreedIndicators = comparisons.filter(c => c.agreement).length
  const disagreedIndicators = evaluatedIndicators - agreedIndicators

  const assessmentScores = await Promise.all(
    assessments.map(async (a) => {
      const totalEvidence = await prisma.fixedEvidence.count()
      const verifiedEvidence = a.evidenceResponses.length
      const score = totalEvidence > 0 ? Math.round((verifiedEvidence / totalEvidence) * 100) : 0
      return { assessorId: a.assessor?.id ?? "unknown", assessorName: a.assessor?.name ?? "غير محدد", score }
    })
  )

  const averageScore = assessmentScores.length > 0
    ? Math.round(assessmentScores.reduce((acc, s) => acc + s.score, 0) / assessmentScores.length)
    : 0

  const domainAverages = domains.map(d => {
    const domainIndicators = comparisons.filter(c => c.domain === d.name && c.evaluatedByAll)
    const domainAgreed = domainIndicators.filter(c => c.agreement).length
    const avg = domainIndicators.length > 0
      ? Math.round((domainAgreed / domainIndicators.length) * 100)
      : 0
    return {
      domain: d.name,
      average: avg,
      band: getPerformanceBand(avg)
    }
  })

  const agreementRate = evaluatedIndicators > 0
    ? Math.round((agreedIndicators / evaluatedIndicators) * 100)
    : 0

  const result: AggregatedResult = {
    totalEvaluations: assessments.length,
    averageScore,
    performanceBand: getPerformanceBand(averageScore),
    domainAverages,
    agreementRate,
    totalIndicators,
    agreedIndicators,
    disagreedIndicators
  }

  return {
    assessments: assessments.map(a => ({
      id: a.id,
      assessorName: a.assessor?.name ?? "غير محدد",
      assessorRole: a.assessor?.role ?? "غير محدد",
      schoolName: a.school?.name || "",
      branchName: a.branch?.name || "",
      status: a.status,
      progress: a.progress,
      startedAt: a.startedAt,
      submittedAt: a.submittedAt
    })),
    comparisons,
    aggregated: result,
    assessmentScores
  }
}

export async function saveEvidenceResponse(assessmentId: string, evidenceId: string, isVerified: boolean, visitNumber: number = 0, scoreValue?: number) {
  const user = await getCurrentUser()
  if (!user) throw new Error("Unauthorized")

  const assessment = await prisma.assessment.findUnique({
    where: { id: assessmentId },
    select: { id: true, assessorId: true, schoolId: true, branchId: true, type: true, status: true }
  })

  if (!assessment) throw new Error("Assessment not found")

  // 1. التحقق من الملكية الصارمة لعزل التقييمات
  if (user.role === "SUPER_ADMIN") {
    if (assessment.type !== "PRIVATE_REVIEW") {
      throw new Error("Forbidden: Super Admin cannot modify operational assessments")
    }
    if (assessment.assessorId !== user.id) {
      throw new Error("Forbidden: You can only edit your own private reviews.")
    }
  } else {
    if (assessment.type === "PRIVATE_REVIEW") {
      throw new Error("Forbidden: Cannot modify private review")
    }
    // منع أي مستخدم من تعديل تقييم يخص مستخدماً آخر
    if (assessment.assessorId !== user.id) {
      throw new Error("Forbidden: You can only modify assessments assigned to you.")
    }
    // فحوصات النطاق الإضافية للأمان
    if (user.role === "SCHOOL_MANAGER" && assessment.schoolId !== user.schoolId) throw new Error("Forbidden: Out of scope")
    if (user.role === "BRANCH_MANAGER" && assessment.branchId !== user.branchId) throw new Error("Forbidden: Out of scope")
  }

  if (isVerified) {
    await prisma.evidenceResponse.upsert({
      where: {
        assessmentId_evidenceId_visitNumber: {
          assessmentId,
          evidenceId,
          visitNumber
        }
      },
      update: { isVerified, scoreValue: scoreValue || null },
      create: {
        assessmentId,
        evidenceId,
        isVerified,
        visitNumber,
        scoreValue: scoreValue || null
      }
    })
  } else {
    await prisma.evidenceResponse.deleteMany({
      where: { assessmentId, evidenceId, visitNumber }
    })
  }

  const totalEvidence = await prisma.fixedEvidence.count()
  const verifiedEvidence = await prisma.evidenceResponse.count({
    where: { assessmentId, isVerified: true }
  })
  
  const progress = totalEvidence > 0 ? Math.round((verifiedEvidence / totalEvidence) * 100) : 0
  
  await prisma.assessment.update({
    where: { id: assessmentId },
    data: { progress, status: "IN_PROGRESS" }
  })
}

  }

  const assessment = await prisma.assessment.findUnique({
    where: { id: assessmentId },
    select: { assessorId: true, schoolId: true, branchId: true, type: true }
  })

  if (!assessment) throw new Error("Assessment not found")

  // التحقق من الملكية الصارمة
  if (assessment.assessorId !== user.id) {
    throw new Error("Forbidden: You can only finish your own assessments.")
  }

  if (user.role === "SCHOOL_MANAGER" && assessment.schoolId !== user.schoolId) throw new Error("Forbidden")
  if (user.role === "BRANCH_MANAGER" && assessment.branchId !== user.branchId) throw new Error("Forbidden")

  const totalIndicators = await prisma.indicator.count({ where: { coverage: "مغطى" } })
  const answeredIndicators = await prisma.evidenceResponse.count({ 
    where: { assessmentId, isVerified: true } 
  })

  if (answeredIndicators < totalIndicators) {
    throw new Error("Cannot finish: There are uncovered indicators remaining.")
  }

  await prisma.assessment.update({
    where: { id: assessmentId },
    data: { 
      status: "COMPLETED", 
      completedAt: new Date(),
      submittedAt: new Date()
    },
  })

  if (assessment.type !== "PRIVATE_REVIEW") {
    const unverifiedEvidence = await prisma.fixedEvidence.findMany({
      where: {
        NOT: {
          responses: {
            some: {
              assessmentId,
              isVerified: true
            }
          }
        }
      },
      include: { indicator: true }
    })

    for (const ev of unverifiedEvidence) {
      const existingGap = await prisma.gap.findFirst({
        where: { assessmentId, indicatorId: ev.indicatorId }
      })

      if (!existingGap) {
        await prisma.gap.create({
          data: {
            schoolId: assessment.schoolId,
            branchId: assessment.branchId,
            assessmentId,
            indicatorId: ev.indicatorId,
            description: `مؤشر لم يثبت تحققه: ${ev.indicator.description}`,
            severity: "HIGH",
            status: "OPEN"
          }
        })
      }
    }

    await prisma.activity.create({
      data: {
        actorId: user.id,
        schoolId: assessment.schoolId,
        branchId: assessment.branchId,
        action: "SUBMIT",
        entityType: "ASSESSMENT",
        entityId: assessmentId,
        metadata: { description: `تم تسليم تقييم بواسطة ${user.name}` }
      }
    })
  }

  revalidatePath("/quality-assessor")
  revalidatePath(`/quality-assessor/assessments/${assessmentId}`)
}

export async function uploadAttachment(assessmentId: string, evidenceId: string, file: { name: string, type: string, size: number, data: string }) {
  const user = await getCurrentUser()
  if (!user) throw new Error("Unauthorized")

  // الإصلاح: كان هذا الإجراء مفتوحاً لأي مستخدم على أي تقييم — الآن يُفرض فحص الكتابة
  const access = await checkAssessmentAccess(assessmentId, "write")
  if (!access.allowed) {
    throw new Error("Forbidden: You can only modify assessments assigned to you.")
  }

  const allowedTypes = ['application/pdf', 'application/msword', 'application/vnd.openxmlformats-officedocument.wordprocessingml.document', 'image/jpeg', 'image/png', 'image/svg+xml']
  if (!allowedTypes.includes(file.type)) throw new Error("نوع الملف غير مدعوم")
  if (file.size > 2 * 1024 * 1024) throw new Error("حجم الملف يتجاوز 2 ميجابايت")

  return await prisma.assessmentAttachment.create({
    data: {
      assessmentId,
      evidenceId,
      uploadedById: user.id,
      fileName: file.name,
      fileType: file.type,
      fileSize: file.size,
      fileData: file.data
    }
  })
}

export async function deleteAttachment(attachmentId: string, assessmentId: string) {
  const user = await getCurrentUser()
  if (!user) throw new Error("Unauthorized")

  // الإصلاح: كان هذا الإجراء يحذف أي مرفق بأي ID بدون أي فحص — ثغرة حذف جماعي
  const attachment = await prisma.assessmentAttachment.findUnique({
    where: { id: attachmentId }
  })
  if (!attachment || attachment.assessmentId !== assessmentId) {
    throw new Error("Attachment not found")
  }

  const access = await checkAssessmentAccess(assessmentId, "write")
  if (!access.allowed) {
    throw new Error("Forbidden: You can only modify assessments assigned to you.")
  }

  return await prisma.assessmentAttachment.delete({
    where: { id: attachmentId }
  })
}

export async function createNewAssessment() {
  const user = await getCurrentUser()
  if (!user || !user.schoolId) throw new Error("Unauthorized")

  const assessment = await prisma.assessment.create({
    data: {
      schoolId: user.schoolId,
      branchId: user.branchId || null,
      assessorId: user.id,
      status: "IN_PROGRESS",
      type: "NORMAL"
    }
  })
  redirect(`/quality-assessor/assessments/${assessment.id}`)
}

export async function getPlatformAnalytics() {
  const [schools, branches, assessments, gaps, totalEvidence, verifiedEvidence] = await Promise.all([
    prisma.school.count(),
    prisma.branch.count(),
    prisma.assessment.findMany({ include: { school: true, branch: true, _count: { select: { gaps: true } } }, orderBy: { startedAt: "desc" }, take: 10 }),
    prisma.gap.findMany({ include: { school: true, branch: true, indicator: true } }),
    prisma.fixedEvidence.count(),
    prisma.evidenceResponse.count({ where: { isVerified: true } })
  ])

  const responseCoverage = totalEvidence > 0 ? Math.round((verifiedEvidence / totalEvidence) * 100) : 0

  return {
    schools,
    branches,
    activeAssessments: assessments.filter(a => a.status === "IN_PROGRESS").length,
    completedAssessments: assessments.filter(a => a.status === "COMPLETED").length,
    recentAssessments: assessments,
    totalGaps: gaps.length,
    openGaps: gaps.filter(g => g.status === "OPEN").length,
    presentCount: verifiedEvidence,
    notPresentCount: gaps.length,
    responseCoverage
  }
}

export async function getSchoolAnalytics(schoolId: string): Promise<BranchAnalytics> {
  const totalEvidence = await prisma.fixedEvidence.count()
  const assessments = await prisma.assessment.findMany({
    where: { schoolId, type: "NORMAL", status: { in: ["IN_PROGRESS", "COMPLETED"] } },
    include: {
      assessor: { select: { id: true, name: true, role: true } },
      branch: { select: { id: true, name: true } },
      evidenceResponses: { where: { isVerified: true } }
    }
  })

  const excludedAssessments = await prisma.assessment.count({
    where: { schoolId, OR: [ { type: "PRIVATE_REVIEW" }, { status: "NOT_STARTED" } ] }
  })

  if (assessments.length === 0) {
    return {
      provisionalPercentage: null,
      finalPercentage: null,
      eligibleAssessments: 0,
      completedAssessments: 0,
      inProgressAssessments: 0,
      excludedAssessments,
      evaluators: [],
      assessments: []
    }
  }

  const mapped: AssessmentScore[] = assessments.map(a => ({
    id: a.id,
    status: a.status,
    score: totalEvidence > 0 ? Math.round((a.evidenceResponses.length / totalEvidence) * 100) : 0,
    evaluatorId: a.assessor?.id ?? "unknown",
    evaluatorName: a.assessor?.name ?? "غير محدد",
    evaluatorRole: a.assessor?.role ?? "غير محدد",
    verifiedCount: a.evidenceResponses.length,
    totalEvidence,
    branchId: a.branchId,
    branchName: a.branch?.name || "كامل المدرسة"
  }))

  const provisionalPercentage = mapped.length > 0
    ? Math.round((mapped.reduce((sum, a) => sum + a.score, 0) / mapped.length) * 10) / 10
    : null

  const completed = mapped.filter(a => a.status === "COMPLETED")
  const finalPercentage = completed.length > 0
    ? Math.round((completed.reduce((sum, a) => sum + a.score, 0) / completed.length) * 10) / 10
    : null

  const evaluatorMap = new Map<string, EvaluatorAnalytics>()
  for (const a of mapped) {
    if (!evaluatorMap.has(a.evaluatorId)) {
      evaluatorMap.set(a.evaluatorId, {
        id: a.evaluatorId,
        name: a.evaluatorName,
        role: a.evaluatorRole,
        assessments: [],
        provisionalAverage: 0,
        finalAverage: null
      })
    }
    evaluatorMap.get(a.evaluatorId)!.assessments.push(a)
  }

  const evaluators: EvaluatorAnalytics[] = Array.from(evaluatorMap.values()).map(ev => {
    const evCompleted = ev.assessments.filter(a => a.status === "COMPLETED")
    ev.provisionalAverage = ev.assessments.length > 0
      ? Math.round((ev.assessments.reduce((sum, a) => sum + a.score, 0) / ev.assessments.length) * 10) / 10
      : null
    ev.finalAverage = evCompleted.length > 0
      ? Math.round((evCompleted.reduce((sum, a) => sum + a.score, 0) / evCompleted.length) * 10) / 10
      : null
    return ev
  })

  return {
    provisionalPercentage,
    finalPercentage,
    eligibleAssessments: mapped.length,
    completedAssessments: completed.length,
    inProgressAssessments: mapped.length - completed.length,
    excludedAssessments,
    evaluators,
    assessments: mapped
  }
}

import { prisma } from "@/lib/prisma"
import { getCurrentUser } from "@/lib/permissions"

export async function savePlan(formData: FormData) {
  const user = await getCurrentUser()
  if (!user || user.role !== "QUALITY_ASSESSOR") throw new Error("Unauthorized: Only Quality Assessors can create plans")

  const id = formData.get("id") as string
  const title = formData.get("title") as string
  const gapId = formData.get("gapId") as string
  const indicatorId = formData.get("indicatorId") as string
  const startDate = formData.get("startDate") as string
  const targetDate = formData.get("targetDate") as string

  if (!title || !targetDate) throw new Error("العنوان وتاريخ الانتهاء مطلوبان")

  const actions: any[] = []
  const entries = Array.from(formData.entries())
  entries.forEach(([key, value]) => {
    const match = key.match(/actions\[(\d+)\]\[(\w+)\]/)
    if (match) {
      const index = parseInt(match[1])
      const field = match[2]
      if (!actions[index]) actions[index] = {}
      actions[index][field] = value
    }
  })

  let schoolId = user.schoolId
  let branchId = user.branchId || null

  if (gapId) {
    const gap = await prisma.gap.findUnique({ where: { id: gapId }, select: { schoolId: true, branchId: true } })
    if (gap) {
      if (!schoolId) schoolId = gap.schoolId
      if (!branchId) branchId = gap.branchId
    }
  }

  if (!schoolId) throw new Error("لم يتم تحديد المدرسة.")

  const data: any = {
    title,
    description: formData.get("description") as string,
    gapId: gapId || null,
    indicatorId: indicatorId || null,
    startDate: new Date(startDate || Date.now()),
    targetDate: new Date(targetDate),
    status: formData.get("status") as string || "DRAFT",
    priority: formData.get("priority") as string || "MEDIUM",
    objective: formData.get("objective") as string,
    reasons: formData.get("reasons") as string,
    strengths: formData.get("strengths") as string,
    improvements: formData.get("improvements") as string,
    baseline: parseFloat(formData.get("baseline") as string) || null,
    targetPercentage: parseFloat(formData.get("targetPercentage") as string) || null,
    currentMeasured: parseFloat(formData.get("currentMeasured") as string) || null,
    kpiName: formData.get("kpiName") as string,
    successCriteria: formData.get("successCriteria") as string,
    requiredSupport: formData.get("requiredSupport") as string,
    followUpMethod: formData.get("followUpMethod") as string,
    impactNotes: formData.get("impactNotes") as string,
    schoolId,
    branchId,
    createdById: user.id,
  }

  try {
    if (id) {
      await prisma.$transaction(async (tx) => {
        await tx.improvementPlan.update({ where: { id }, data })
        await tx.improvementAction.deleteMany({ where: { planId: id } })
        if (actions.length > 0) {
          await tx.improvementAction.createMany({
            data: actions.map(a => ({
              planId: id,
              title: a.title,
              status: a.status || "PENDING",
              startDate: a.startDate ? new Date(a.startDate) : null,
              dueDate: a.dueDate ? new Date(a.dueDate) : null,
            }))
          })
        }
      })
    } else {
      await prisma.$transaction(async (tx) => {
        const plan = await tx.improvementPlan.create({ data })
        if (actions.length > 0) {
          await tx.improvementAction.createMany({
            data: actions.map(a => ({
              planId: plan.id,
              title: a.title,
              status: a.status || "PENDING",
              startDate: a.startDate ? new Date(a.startDate) : null,
              dueDate: a.dueDate ? new Date(a.dueDate) : null,
            }))
          })
        }
      })
    }

    revalidatePath("/school-manager/quality/improvement-plans")
    revalidatePath("/improvement-plans")
    return { success: true }
  } catch (error) {
    console.error(error)
    throw new Error("حدث خطأ أثناء حفظ الخطة")
  }
}

// تحديث نسبة تقدم الإجراء (مع التحقق من التزامن بين الحالة والنسبة)
export async function updateActionProgress(actionId: string, status: string, planId: string, progressInput: number = 0) {
  const user = await getCurrentUser()
  if (!user || user.role !== "QUALITY_ASSESSOR") throw new Error("Unauthorized")

  let finalProgress = progressInput;

  // 1. فرض القواعد من جانب الخادم لضمان عدم وجود تضارب
  if (status === "COMPLETED") {
    finalProgress = 100;
  } else if (status === "PENDING") {
    finalProgress = 0;
  } else {
    // للحالات (جارية) و (متعثرة): النسبة يجب أن تكون بين 0 و 99
    finalProgress = Math.max(0, Math.min(99, finalProgress));
    // إذا أدخل المستخدم 100% للحالة جارية، يتم خفضها إلى 99% تلقائياً
    if (finalProgress === 100) finalProgress = 99;
  }

  // 2. تحديث الإجراء بالحالة والنسبة الجديدة المعتمدة
  await prisma.improvementAction.update({
    where: { id: actionId },
    data: { 
      status,
      progress: finalProgress
    }
  })

  // 3. إعادة حساب النسبة الإجمالية للخطة
  const actions = await prisma.improvementAction.findMany({ where: { planId } })
  
  if (actions.length > 0) {
    const totalProgress = actions.reduce((acc, a) => acc + (a.progress || 0), 0)
    const overallProgress = Math.round(totalProgress / actions.length)
    
    await prisma.improvementPlan.update({
      where: { id: planId },
      data: { progress: overallProgress }
    })
  } else {
    await prisma.improvementPlan.update({
      where: { id: planId },
      data: { progress: 0 }
    })
  }

  revalidatePath(`/school-manager/quality/improvement-plans/${planId}`)
}

export interface PerformanceLevel {
  label: string
  color: string
  level: number
}

export function getPerformanceLevel(percentage: number | null | undefined): PerformanceLevel {
  if (percentage === null || percentage === undefined || isNaN(percentage as number)) {
    return { label: 'غير مقيم', color: '#858B95', level: 0 }
  }
  if (percentage >= 90) return { label: 'التميز', color: '#16A34A', level: 4 }
  if (percentage >= 75) return { label: 'التقدم', color: '#2878B8', level: 3 }
  if (percentage >= 60) return { label: 'الانطلاق', color: '#E9A34A', level: 2 }
  if (percentage >= 50) return { label: 'التهيئة', color: '#C62828', level: 1 }
  return { label: 'غير مقيم', color: '#858B95', level: 0 }
}
