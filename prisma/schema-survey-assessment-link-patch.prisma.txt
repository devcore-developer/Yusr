# ترقيع السكيما — ربط الاستبانات بالتقييم (assessment-specific surveys)
# إضافي بحت (additive-only): عمود nullable واحد + علاقة FK + فهرس.
# لا حذف سجلات، لا تعديل أعمدة قائمة، لا فقدان بيانات —
# الاستبانات التاريخية (بلا تقييم) تنجو كما هي وتظهر «غير مرتبطة بتقييم (سجل تاريخي)»،
# ولا يُربط أي سجل قديم بصمت بأي تقييم (قرار التصميم الملزم للمهمة).
# ============================================================

# ---------- 1) نموذج Survey القائم — أضف داخل التعريف: ----------
  assessmentId String?

  assessment   Assessment? @relation(fields: [assessmentId], references: [id], onDelete: SetNull)

  @@index([assessmentId])

# ---------- 2) نموذج Assessment القائم — أضف داخل التعريف سطرًا واحدًا: ----------
  surveys Survey[]
